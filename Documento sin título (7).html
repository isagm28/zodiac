class QuizApp {

constructor() {

this.currentStep = 1;

this.currentQuestionIdx = 0;

this.step1Answers = [];

this.step2Answers = [];

this.dominantType = 'A';

this.exactSignName = "Aries";

// DOM Bindings

this.welcomeView = document.getElementById('welcome-view');

this.quizView = document.getElementById('quiz-view');

this.step2IntroView = document.getElementById('step2-intro-view');

this.resultView = document.getElementById('result-view');

this.progressBar = document.getElementById('progress-bar');

this.progressText = document.getElementById('progress-text');

this.questionText = document.getElementById('question-text');

this.optionsContainer = document.getElementById('options-container');

this.stepBadge = document.getElementById('step-badge');

this.questionCounter = document.getElementById('question-counter');

this.prevBtn = document.getElementById('prev-btn');

this.nextBtn = document.getElementById('next-btn');

}

startQuiz() {

this.welcomeView.classList.add('hidden');

this.quizView.classList.remove('hidden');

this.currentStep = 1;

this.currentQuestionIdx = 0;

this.step1Answers = new Array(QUIZ_DATA.generalQuestions.length).fill(null);

this.renderQuestion();

}

renderQuestion() {

const currentQuestions = this.currentStep === 1

? QUIZ_DATA.generalQuestions

: QUIZ_DATA.step2Modules[this.dominantType].questions;

const qData = currentQuestions[this.currentQuestionIdx];

// Progress Bar

let progressPercent = 0;

if (this.currentStep === 1) {

progressPercent = Math.round(((this.currentQuestionIdx) / 21) * 100);

this.stepBadge.textContent = "Step 1: Primary Archetype";

this.questionCounter.textContent = `Question ${this.currentQuestionIdx + 1} of
17`;

} else {

progressPercent = Math.round(((17 + this.currentQuestionIdx) / 21) * 100);

this.stepBadge.textContent = "Step 2: Zodiac Sub-Quiz";

this.questionCounter.textContent = `Question ${this.currentQuestionIdx + 1} of
4`;

}

this.progressBar.style.width = `${progressPercent}%`;

this.progressText.textContent = `${progressPercent}%`;

// Question Title

this.questionText.textContent = qData.q;

// Render Options

this.optionsContainer.innerHTML = '';

const selectedAns = this.currentStep === 1

? this.step1Answers[this.currentQuestionIdx]

: this.step2Answers[this.currentQuestionIdx];

qData.options.forEach((opt, idx) => {

const isSelected = selectedAns === opt.type;

const letter = String.fromCharCode(65 + idx);

const optBtn = document.createElement('button');

optBtn.className = `option-card w-full text-left p-4 sm:p-5 rounded-2xl border-2
font-medium flex items-center gap-4 transition-all ${

isSelected

? 'selected border-buzz bg-pink-50/60 text-slate-900 shadow-sm'

: 'border-slate-200 hover:border-slate-300 bg-slate-50/50 text-slate-700'

}`;

optBtn.innerHTML = `

<span class="w-8 h-8 rounded-xl flex items-center justify-center font-display
font-bold text-sm ${

isSelected ? 'bg-buzz text-white' : 'bg-slate-200 text-slate-700'

}">${letter}</span>

<span class="flex-1 text-sm sm:text-base">${opt.text}</span>

${isSelected ? '<i class="fa-solid fa-circle-check text-buzz text-xl"></i>' : ''}

`;

optBtn.onclick = () => this.selectOption(opt.type);

this.optionsContainer.appendChild(optBtn);

});

// Prev/Next Navigation

this.prevBtn.disabled = (this.currentStep === 1 && this.currentQuestionIdx === 0);

this.nextBtn.disabled = (selectedAns === undefined || selectedAns === null);

if (this.currentQuestionIdx === currentQuestions.length - 1) {

this.nextBtn.innerHTML = this.currentStep === 1

? 'Finish Step 1 <i class="fa-solid fa-arrow-right ml-1"></i>'

: 'See Final Results <i class="fa-solid fa-sparkles ml-1"></i>';

} else {

this.nextBtn.innerHTML = 'Next Question <i class="fa-solid fa-arrow-right
ml-1"></i>';

}

}

selectOption(type) {

if (this.currentStep === 1) {

this.step1Answers[this.currentQuestionIdx] = type;

} else {

this.step2Answers[this.currentQuestionIdx] = type;

}

this.renderQuestion();

}

nextQuestion() {

const currentQuestions = this.currentStep === 1

? QUIZ_DATA.generalQuestions

: QUIZ_DATA.step2Modules[this.dominantType].questions;

if (this.currentQuestionIdx < currentQuestions.length - 1) {

this.currentQuestionIdx++;

this.renderQuestion();

} else {

if (this.currentStep === 1) {

this.processStep1Results();

} else {

this.processStep2Results();

}

}

}

prevQuestion() {

if (this.currentQuestionIdx > 0) {

this.currentQuestionIdx--;

this.renderQuestion();

} else if (this.currentStep === 2) {

this.currentStep = 1;

this.currentQuestionIdx = QUIZ_DATA.generalQuestions.length - 1;

this.renderQuestion();

}

}

processStep1Results() {

const counts = { A: 0, B: 0, C: 0, D: 0 };

this.step1Answers.forEach(ans => {

if (counts[ans] !== undefined) counts[ans]++;

});

let maxCount = -1;

let dominant = 'A';

for (const key in counts) {

if (counts[key] > maxCount) {

maxCount = counts[key];

dominant = key;

}

}

this.dominantType = dominant;

const module = QUIZ_DATA.step2Modules[dominant];

document.getElementById('element-badge-icon').textContent =
QUIZ_DATA.archetypes[dominant].emoji;

document.getElementById('step2-title').textContent = `You are
${QUIZ_DATA.archetypes[dominant].title}!`;

document.getElementById('step2-description').textContent = `Now answer 4
targeted sub-questions in the ${module.name} module to discover your exact Zodiac sign!`;

this.quizView.classList.add('hidden');

this.step2IntroView.classList.remove('hidden');

}

startStep2() {

this.step2IntroView.classList.add('hidden');

this.quizView.classList.remove('hidden');

this.currentStep = 2;

this.currentQuestionIdx = 0;

this.step2Answers = new Array(4).fill(null);

this.renderQuestion();

}

processStep2Results() {

const counts = { A: 0, B: 0, C: 0 };

this.step2Answers.forEach(ans => {

if (counts[ans] !== undefined) counts[ans]++;

});

let maxCount = -1;

let dominantSignKey = 'A';

for (const key in counts) {

if (counts[key] > maxCount) {

maxCount = counts[key];

dominantSignKey = key;

}

}

const module = QUIZ_DATA.step2Modules[this.dominantType];

this.exactSignName = module.signs[dominantSignKey];

this.renderFinalResult();

}

renderFinalResult() {

this.quizView.classList.add('hidden');

this.resultView.classList.remove('hidden');

this.progressBar.style.width = '100%';

this.progressText.textContent = '100%';

const archetype = QUIZ_DATA.archetypes[this.dominantType];

const cardContainer = document.getElementById('archetype-card');

cardContainer.className = `rounded-2xl p-6 mb-8 border transition-all
${archetype.bgClass}`;

// Fill Primary Archetype

document.getElementById('result-emoji').textContent = archetype.emoji;

document.getElementById('result-title').textContent = archetype.title;

document.getElementById('result-motto').textContent = archetype.motto;

document.getElementById('result-description').textContent =
archetype.description;

document.getElementById('result-weakness').textContent = archetype.weakness;

// Fill Document 6 Zodiac Data

const zodiacData = QUIZ_DATA.zodiacDescriptions[this.exactSignName];

document.getElementById('zodiac-symbol').textContent = zodiacData.symbol;

document.getElementById('zodiac-title').textContent = zodiacData.title;

const descContainer = document.getElementById('zodiac-full-description');

descContainer.innerHTML = '';

zodiacData.paragraphs.forEach(pText => {

const p = document.createElement('p');

p.textContent = pText;

descContainer.appendChild(p);

});

const now = new Date();

document.getElementById('export-date').textContent = `Issued:
${now.toLocaleDateString()}`;

window.scrollTo({ top: 0, behavior: 'smooth' });

}

exportPDF() {

const element = document.getElementById('pdf-export-container');

const opt = {

margin:
[0.4, 0.4, 0.4, 0.4],

filename:
`Quiz_Result_${this.exactSignName}.pdf`,

image:
{ type: 'jpeg', quality: 0.98 },

html2canvas: { scale: 2, useCORS: true },

jsPDF:
{ unit: 'in', format: 'letter', orientation: 'portrait' }

};

html2pdf().set(opt).from(element).save();

}

resetQuiz() {

this.currentStep = 1;

this.currentQuestionIdx = 0;

this.step1Answers = [];

this.step2Answers = [];

this.resultView.classList.add('hidden');

this.welcomeView.classList.remove('hidden');

this.progressBar.style.width = '0%';

this.progressText.textContent = '0%';

window.scrollTo({ top: 0, behavior: 'smooth' });

}

}

// Initialize App on Window Load

let quizApp;

window.onload = function() {

quizApp = new QuizApp();

};

</script>
